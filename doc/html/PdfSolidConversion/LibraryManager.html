def initialize_sdk(resource_path = nil)
  path = resource_path.nil? || resource_path.to_s.empty? ? default_resource_path : resource_path.to_s
  fonts_path = File.join(path, "fonts")
  unless File.directory?(fonts_path)
    raise ArgumentError, "SDK resource fonts directory not found: #{fonts_path}"
  end

  Native.CPDF_Initialize(path)
  nil
end
